#Como "esconder" um atributo?
#Motivação : As vezes, há certos atributos que não queremos que sejam alterados pelo usuário diretamente.
#Ao invés disso, normalmente tais atributos são alterados através de métodos.
#Para isso, basta colocar UM underline antes do nome da variável.
#Acessar diretamente significa acessar o atributo SEM utilizar um metodo como intermediario.
#Portanto, dentro do metodo é possível acessar.
#Exemplo:
def __init__(self, nivel = 100):
    self.variavel_1 = 100
    self._nome_variavel = nivel #Atributo "escondido".

#Atenção : Neste tipo, é possível acessar diretamente ao fazer, por exemplo, self._nome_variavel em qualquer momento.


#Atributo "duplamente escondido" : Atributo cujo nome tem DOIS underlines como prefixo.
#Exemplo:
__variavel = 100 #"Duplamente escondido" / privado
#Diferentemente do anterior (com UM underline), não é possível acessar diretamente DESTA FORMA. Ou seja:
#objeto.__variavel resulta em erro, a menos que seja acessado via metodo.
#Entretanto, tem um equivalente.
# objeto.__variavel é reconhecido no Python como objeto._NomeClasse_NomeAtributo
#Portanto, é possível acessar diretamente ao fazer objeto._NomeClasse_NomeAtributo

#Portanto, toda vez que se faz instancia_objeto.metodo(x,y,z), é como se chamasse:
#       instancia_objeto.metodo(ref_objeto (self) , x, y, z)
#Isso signfica que fazer instancia_objeto.metodo(instancia_objeto, x, y,z) resulta em ERRO.
#Motivo : O Python ja passa, implicitamente, o objeto instanciado como primeiro parametro.

#__eq__
#Afeta comperaçãoes. Por exemplo, como reagir ao comparar objeto1 == objeto2 (mesmas classe)
#Isto é necessário, pois caso o contrário, mesmo se os objeos tenham exatamente os mesmos atributos (e valores)
#o python compararia as posições na memória.
#Exemplo
class Posicao:

    def __eq__(self, outraPosicao):
        if not isinstance(outraPosicao, Posicao): #Verifica se é uma instancia desta classe.
            return False
        return self.x == outraPosicao.x and self.y == outraPosicao.y

p1 = Posicao(3,4)
p2 = Posicao(3,4)
p1 == p2 #Posicao.__eq__(p1, p2)#Sem o override do __eq__, iria dar false, pois compararia memoria.

# Caso exista um dicionario, lista, etc que o objeto deva ter, NUNCA use-os com valores iniciais.
# Motivo : Aquele mesmo esquema de pegadinha de parâmetros de funções, pois o python cria esta lista
# dicionario, etc e, desta forma, toda vez que chamar a função irá passar o mesmo objeto (lista, dict, etc)
# Portanto, qualquer alteração em um objeto afeta o outro.

#Exemplo de NÃO FAZER

    def __init__(self, lista_obstaculos = []):
        pass

robo1.lista_obstaculos.append(X) // Ou qualquer outra manipulação

#Resultado final : A propriedade lista_obstaculos em ambos os objetos (robo1, robo2) apontam
#para a mesma lista e, portanto, alterar em um afeta o outro.
#Isto NORMALMENTE é indesejado.


#Exemplo de COMO FAZER (boas práticas)
class Robo:
def __init__(self, lista_obstaculos = None):
    #E, aqui dentro, verificar se é None. Se for None, cria a lista.
    if lista_obstaculos == None:
        #Cria
        self.lista_obstaculos = []
    else:
        #Se o que foi recebido NÃO é vazio, atribua a ele.
        self.lista_obstaculos = lista_obstaculos
    #Equivalente a self.obstaculos = obstaculos if obstaculos is not None else []
#Desta forma, os objetos CRIADOS não compartilharão referências de lista, dicts, etc.
#Lembre-se que isso NÃO acontece para os outros tipos (inteiros, string, etc)


#Sobre atributos escondidos (_nome_atributo)
#Exemplo:
self.nome_variavel = 100 #É possível acessar diretamente
self._nome_variavel = 100 #É possível acessar diretamente, mas pela convenção, entende-se que não deve.
self.__nome_variavel = 100 #NÃO É POSSÍVEL ACESSAR DIRETAMENTE, exceto caso seja acessado via um método.
#Ao tentar acessar DIRETAMENTE a variável (fora de metodos), o python retornará um erro.
#Dizendo que não encontrou tal variavel.
